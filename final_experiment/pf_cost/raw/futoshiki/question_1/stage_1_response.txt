# 工具函数：将 Z3 解模型中的 cell[(r,c)] 提取为二维矩阵，并格式化输出

from typing import List, Any

def to_grid(n: int, rows: List[int], cols: List[int], cell: Any, model: Any):
    """
    将 Z3 model 中的 cell[(r,c)] 提取为二维 Python 列表
    参数:
      - n: 阶数
      - rows/cols: 索引列表
      - cell: CustomSym 字典式符号容器（keys: (r,c)）
      - model: Z3 Model (PuzzleClone 中的 _solutions[0])
    返回:
      - grid: List[List[int]]
    """
    grid = []
    for r in rows:
        row = []
        for c in cols:
            v = model.evaluate(cell[(r, c)], model_completion=True)
            row.append(int(str(v)))
        grid.append(row)
    return grid

def format_grid(grid):
    """
    按题目要求输出嵌套列表的字符串形式，如：
    [[4, 3, 2, 1],[2, 1, 4, 3],[1, 4, 3, 2],[3, 2, 1, 4]]
    """
    return "[" + ",".join(["[" + ", ".join(str(x) for x in row) + "]" for row in grid]) + "]"
