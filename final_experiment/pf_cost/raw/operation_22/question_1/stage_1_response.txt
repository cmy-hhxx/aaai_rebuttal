# pf_operation_22_0.py
import sympy as sp

x, y = sp.symbols('x y')

def pf_op(f_expr: str, g_expr: str):
    """
    自定义运算：f■g = ∂f/∂x + ∂g/∂x
    输入为可被 sympy.sympify 解析的字符串表达式。
    返回 sympy 表达式。
    """
    f = sp.sympify(f_expr)
    g = sp.sympify(g_expr)
    return sp.diff(f, x) + sp.diff(g, x)

def sx_to_latex(expr):
    """将 sympy 表达式转为 LaTeX 字符串（不带 $ 符号）。"""
    return sp.latex(sp.simplify(expr))
