# pf_skyscraper_puzzle_0.py
# 用于 4x4（及一般 n×n）摩天楼谜题的“可见楼数”计算
from z3 import If, Sum

def visible_count(seq):
    """
    从左到右计算可见楼数（Z3 友好版本）。
    可见规则：从观察方向起，若当前楼高度严格大于此前最高楼，则可见数+1，并更新最高楼。
    seq: List[z3.ArithRef] / List[int] 均可（主要给 Z3 用）
    return: z3.ArithRef（可用于等式约束）
    """
    max_h = 0
    terms = []
    for x in seq:
        seen = If(x > max_h, 1, 0)
        terms.append(seen)
        # 更新 max_h（使用 If，保持表达式可符号化）
        max_h = If(x > max_h, x, max_h)
    return Sum(terms)

def visible_count_rev(seq):
    """
    从右到左（或从下到上）计算可见楼数：等价于对序列反向后调用 visible_count。
    """
    return visible_count(list(reversed(seq)))
